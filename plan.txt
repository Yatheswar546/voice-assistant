Change these values if tracking after Jan 1, 2027.COST_PER_1M_INPUT_INR = 0.75 * 84.0   # ~63 INR per million input tokensCOST_PER_1M_OUTPUT_INR = 3.75 * 84.0  # ~315 INR per million output tokens
def estimate_cost_inr(input_tokens: int, output_tokens: int) -> float:
    """Calculates the estimated cost of the turn in Indian Rupees (INR)"""
    input_cost = (input_tokens / 1_000_000) * COST_PER_1M_INPUT_INR
    output_cost = (output_tokens / 1_000_000) * COST_PER_1M_OUTPUT_INR
    return input_cost + output_cost
def run_chatbot():
    # 3. Start a continuous chat session to preserve conversation history
    print(f"Initializing Chat Session with {MODEL_NAME}...")
    chat = client.chats.create(model=MODEL_NAME)
    
    total_session_cost_inr = 0.0
    print("\n=======================================================")
    print("Gemini Flash Chatbot with Live Wallet Tracking Initialized!")
    print("Type 'exit' or 'quit' to end the session.")
    print("=======================================================\n")

    while True:
        try:
            user_input = input("You: ").strip()
            if not user_input:
                continue
            if user_input.lower() in ['exit', 'quit']:
                print("\nEnding session.")
                print(f"Total Session Wallet Consumption: ₹{total_session_cost_inr:.4f}")
                break

            # 4. Send message to the model
            # Note: We request usage metadata explicitly to track tokens
            response = chat.send_message(
                user_input,
                config=types.GenerateContentConfig(
                    temperature=0.7,
                )
            )

            # 5. Extract token statistics from the response metadata
            usage = response.usage_metadata
            input_tokens = usage.prompt_token_count
            output_tokens = usage.candidates_token_count
            
            # Calculate metrics
            turn_cost = estimate_cost_inr(input_tokens, output_tokens)
            total_session_cost_inr += turn_cost

            # 6. Display Response and Cost Data
            print(f"\nGemini: {response.text}")
            print("-" * 40)
            print(f"Tokens Used -> Input: {input_tokens} | Output: {output_tokens}")
            print(f"Cost for this turn: ~₹{turn_cost:.4f}")
            print(f"Total session drain: ~₹{total_session_cost_inr:.4f}")
            print("=" * 40 + "\n")

        except Exception as e:
            print(f"\nAn error occurred: {e}\n")
if __name__ == "__main__":
    run_chatbot()
